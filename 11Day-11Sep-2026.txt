---
- name: install helm
  hosts: devops-server
  become: yes
  tasks: 
    - name: download helm
      get-url: 
        url: https://get.helm.sh/helm-v3.7.0-linux-amd.tar.gz
        dest: /tmp/helm.tar.gz
    - name: Exract helm
      unarchive:
        src: /tmp/helm.tar.gz
        dest: /usr/local/bin/
        remote_src: yes
    - name: set helm binary permission
      file: 
        path: /usr/local/bin/helm
        mode: '0755'




resource  "aws_eip" "devops_eip" {
  instance  = aws_instance.devops_ec2.id
}

resource  "aws_instance" "devops_ec2" {
  ami                    = "ami-abcd1234abcd1234"
  instance_type          = "t3.medium"
  
  tags = {
    Name = "devops-ec2"
  }
}






pipeline {
    agent server 
    enviroment { 
        IMAGE_NAME = 'devops-web-app'
    }
    steges { 
        stage('Checkout') {
            steps {
                git 'https://github.com/ravibhadarge/sapmle-repo.git'
             }
          }
          stage('Build Image') {
              steps {
                  script {
                      docker.build(${env.IMAGE_NAME}")
                  }
              }
          }   
          stage('Push Image') {
              steps {
                  script {
                      docker.withRegistry(http://docker.io, 'docker-credentials') {
                        docker.image("${env.IMAGE_NAME}").push('latest')    
                      } 
                  }
              }
          }    
          stage('deploy to k8s')
              steps {
                  script {
                      sh 'kubectl apply -f deployment.yml'
                  }
              }
          }
      }
 }



stages:
  - build
  - push
  - deploy

build:
  stage: build
  script:
    - docker build -t devops-web-app .

push:
  stage: push
  script:
    - docker login -u "$CI_REGISTRY_USER" -p "$CI_REGISTRY_PASSWORD"
    - docker push devops-web-app


deploy:
  stage: deploy
  script:
    - helm repo add stable https://charts.helm.sh/stable
    - helm install devops-nginx stable/nginx-ingress
