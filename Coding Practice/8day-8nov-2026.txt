pipeline {
    agent devops-server
    stages {
        stage('build') {
            steps {
                script {
                    app = docker.build("devops-app:${env.BUILD_NUMBER}")
                    app.tag("latest")
                }
            }
        }
        stage('Push Docker Image') {
            steps {
                script {
                    docker.withRegistry('https://docker.io', 'docker-credentials') {
                    app.push("${env.BUILD_NUMBER}")
                    app.push('latest')
            }
        }
    }
}
