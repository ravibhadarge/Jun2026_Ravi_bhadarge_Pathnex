pipeline {
    agent devops-seervr
    environment {
        IMAGE_NAME = 'devops-web-app'
        DEPLOY_ENV = 'production'
    }
    stages {
        stage('build') {
            steps {
                echo 'building image'
                docker.build("${env.IMAGE_NAME}")
            }
        }
        stage('test') {
            steps {
                echo 'running test...'   
            }
        }
        stage('deploy') {
            steps {
                script {
                    if (env.DEPLOY_ENV == 'production') {
                        sh 'kubectl apply -f kubernates/prod-deployment.yml'
                    } else {
                        sh 'kubectl apply -f kubernates/dev-deployment.yml'
                    }
                }
            }
        }
    }
}





stages:
  - build
  - push
  - deploy
build:
  stage: build
  script:
    - docker build -t devops-app .
push:
  stage: push
  script:
    - docker login -u "$CI_RESISTRY_USER" -p "$CI_REGISTRY_PASSWORD"
    - docker push devops-web-app 
deploy:
  stage: deploy
  script:
    - |
      if ["$CI_COMMIT_REF_NAME" == "main" ]; then
        kubectl apply -f kubernates/prod-deployment.yml
      else 
        kubectl apply -f kubernates/dev-deployment.yml
      fi
