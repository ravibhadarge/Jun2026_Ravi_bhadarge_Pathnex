---
- name: configure firewall with iptable
  hosts: devops-server
  become: yes
  tasks:
    - name: set defualt policies to drop all traffic
      command: iptable -P INPUT DROP
    - name: allow SSH and HTTP traffic
      command: iptable -A INPUT -p tcp --dport 22 -j ACCEPT
      command: iptable -A INPUT -p tcp --dport 80 -j ACCEPT
    - name: save iptable configuration
      command: service iptables save
 



resource "aws_security_group" "devops_sg" {
  name               = "devops-sg-"
  description        = "allow SSH and HTTP"
  ingress {
    from_port        = 22
    to_port          = 22
    protocol         = tcp
    cidr_block       = ["0.0.0.0/0"]
  }
  ingress {
    from_port        = 80
    to_port          = 80
    protocol         = tcp
    cidr_block       = ["0.0.0.0/0"]
  }
  egress {
    from_port        = 0
    to_port          = 0 
    protocol         = "-1"
    cidr_block       = ["0.0.0.0/0"]
  }

                        docker.image('devops-web-app').push('letest')
                    }
                }
            }
        }     
    }
}
