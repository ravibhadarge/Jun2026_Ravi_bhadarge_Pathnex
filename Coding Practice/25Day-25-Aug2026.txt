resource "aws_cloudewatch_metric_alarm"  "dervserver-_cpu_alarm"
  alarm_name              = "deveserver-cpu-utilization-high"
  comparison_operator     = "GreaterThenThreshold"
  evaluation_peroids      = "1"
  metric_name             = "CPUUtilization"
  namespace               = "AWS/EC2"
  period                  = "60"
  statistic               = "Average"
  threshold               = "75"
  
  dimensions = {
    InstanceId  = aws_instance.deveserver_ec2.id
  }

  alarm_action  = [aws_sns_topic.devserver_alert.arn]
 }
