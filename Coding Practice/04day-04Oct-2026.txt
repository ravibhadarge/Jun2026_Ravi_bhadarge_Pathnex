---
- name: install and configure nginx
  hosts: web-server
  become: yes
  tasks:
    - name: install nginx
      yum:
        name: nginx
        state: present
    - name: configure nginx
      template:
        src: nginx.conf.j2
        dest: /etc/nginx/nginx.conf
    - name: start nginx service
      service: 
        name: nginx
        stated: started
        enabled: yes
