--- 
- name: setup nginx with revers proxy
  hosts: prod-server
  become: yes
  tasks: 
    - name: install nginx
      yum: 
        name: nginx
        state: present
    - name: configure nginx revers proxy
      template:
        src: reverse-proxy.conf.j2
        dest: /etc/nginx/conf.d/reverse-proxy.conf
    - name: restart nginx service
      service:
        name: nginx
        state: restarted 
        enabled: yes
