- name: implement the ssh key authonticatation
  hosts: ngnix-server
  become: yes
  tasks:
    - name: Generate ssh key pair
      command: ssh-keygen -t rsa -b 2048 -f /home/ravi/.ssh/id_rsa -N ""
    - name: add ssh public key to authorized_key
      authorized_key:
        user: ravi
        state: present 
        key: "{{ lookup('file', '/home/ravi/.ssh/id_rsa.pub')}}"

apiVersion: networking.k8s.io/v1
kind: NetworkPolicy
metadata:
  name: online-shopping-web-policy
spec:
  podSelector:
    matchLabels:
      app: online-shopping
  policyTypes:
    - Ingress
    - Egress
  ingress:
    - from:
        - podSelector:
            matchLabels:
              app: ravi-api
  egress:
    - to:
        - podSelector:
            matchLabels:
              app: ravi-database



 agent nginx-server
    environment {
        AWS_ACCESS_KEY_ID = credentials('aws-access-key')
        AWS_SECRET_ACCESS_KEY = credentials('aws-secret-key')
    }
    stages {
        stage('Build') {
            steps {
                echo 'Building Docker image...'
                docker.build('pathnex-web-app')
            }
        }
        stage('Deploy') {
            steps {
                echo 'Deploying to AWS EC2...'
                sh 'aws ec2 run-instances --image-id ami-0abcd1234 --count 1 --instance-type t2.micro'
            }
        }
    }
}


stages:
  - build
  - deploy

build:
  stage: build
  script:
    - docker build -t online-shopping-app .

deploy:
  stage: deploy
  script:
    - echo "Deploying with secure credentials"
    - aws ec2 run-instances --image-id ami-0abcd1234 --count 1 --instance-type t2.micro --access-key $AWS_ACCESS_KEY --secret-key $AWS_SECRET_KEY
