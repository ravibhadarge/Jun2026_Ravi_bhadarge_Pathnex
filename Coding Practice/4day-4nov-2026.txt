---
- name: install and configure nginx
  hosts: dev-server
  become: yes
  tasks:
    - name: install nginx
      yum:
        name: nginx
        state: present 
    - name: configur nginx
      template:
        src: ngnix.conf.j2
        dest: /var/www/html/nginx.conf    
    - name: nginx server restart
      service: 
        name: nginx
        state: restarted
        enabled: yes





resource "aws_security_group" "devops_security"
  name               = "devops_security_group"
  description        = "allow for ssh and http"
  ingress {
    from_port  = 22
    to_port    = 22
    protocol   = tcp
    cidr_block = ["0.0.0.0/0"]
  }
  ingress {
    from_port  = 80
    to_port    = 80
    protocol   = tcp
    cidr_block = ["0.0.0.0/0"]
  }
  egress {
    from_port  = 0
    to_port    = 0
    protocol   = "-1"
    cidr_block = ["0.0.0.0/0"]
  }
}
