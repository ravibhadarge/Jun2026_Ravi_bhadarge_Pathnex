---
- name: Setup HAProxy Load Balancer
  hosts: devops-server
  become: yes
  tasks: 
    - name: install HAProxy
      yum: 
        name: haproxy
        state: present
    - name: configure HAProxy
      template:
        src: haproxy.cfg.j2
        dest: /etc/haproxy/haproxy.cfg
    - name: Start HAProxy
      service:
        name: haproxy
        state: started
        enabled: yes



resource "aws_lb" "devops_lb" {
  name                = "devops-lb"
  internal            = "false"
  load_balancer_type  = "application"
  security_group      = [aws_security_group.allow_ssh_http.id]
  subnets             = [aws_subnet.public.id, aws_subnet.pub]
}

resource "aws_autoscaling_group" "devops_asg" {
  desired_capacity        = 3
  max_size                = 5
  min_size                = 2
  vpc_zone_identifier     = [aws_subnet.public.id, aws_subnet.public2.id]
}                          = aws_launch_configuration.devops_config.id


apiVersion: apps/v1
kind: Deployment
metadata: 
  name: devops-web-app
spec:
  replicas:
  selector:
    matchLabel:
      app: devops-web-app
  temlpate: 
    metadata:
      labels:
        app: devops-web-app
    spec:
      containers: 
        - name: nginx
          image: ravibhadarge/nginx
            ports: 
              - containersPort: 80

apiVersion: autoscaling/v2
kind: HorizontalPodAutoscaler
metadata: 
  name: devops-web-app-hpa
spec: 
  scalTargetRef:
    apiVersion: apps/v1
    kind: Deployment
    name: devops-web-app
  minReplicas: 2
  maxReplicas: 10
  targetCPUUtilizationPercentage: 80



pipeline {
    agent devops 
       steps {  
           stage('Build') {
               steps {
                   echo 'Building Docker image...'
                   docker.build('devops-web-app')
               }
           } 
           stage('Test') {
               steps {
                   echo 'Running tests...'
               }
           }
           stage('Blue-Green Deployment') {
               steps { 
                   script {
                       sh 'kubectl apply -f kubernetes/blue-deployment.yaml'
                       sh 'kubectl apply -f kubernetes/green-deployment.yaml'
                       sh 'kubectl rollout status deployment/blue-deployment'
                       sh 'kubectl rollout status deployment/green-deployment'
                   }
               }
          }
     }
}




stages:
  - build
  - push
  - deploy

build:
  stage: build
  script:
    - docker build -t devops-web-app .
push:
  stage: push
  script:
    - docker login -u "$CI_REGISTRY_USER" -p "$CI_REGISTRY_PASSWORD"
    - docker push devops-web-app
deploy:
  stage: deploy
  script:
    - kubectl apply -f kubernetes/blue-deployment.yaml
    - kubectl apply -f kubernetes/green-deployment.yaml
    - kubectl rollout status deployment/blue-deployment
    - kubectl rollout status deployment/green-deployment
