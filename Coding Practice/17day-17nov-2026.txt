- name: configure auto-healing for web server
  hosts: devops-server
  become: yes
  tasks: 
    - name: ensure ngnix service is running
      service:
        name: nginx
        stage: started
        enabled: yes
    - name: setup aut-restart for ngnix service
      lineinfile:
        path: /etc/ststemd/system/nginx.service.d/override.conf
        line: "Restart=always"
        create: yes
    - name: Reload systemd
      command: systemctl daemon-reload
