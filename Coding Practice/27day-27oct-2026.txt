- name: deploy serverless Lambda Function
  hosts: localhost
  tasks:
    - name: create lambda function
      command: aws lambda create-function --function-name devops-lambda --runtime nodejs14.x --role arn:aws:iam::123456789012:role/lambda-role --handler index.handler --zip-file fileb://lambda.zip



resource "aws_dynamodb_table" "devops_table" {
  name             = "devops"
  has_key          = "id"
  billing_mode     = "PAY_PER_REQUEST"
  attribute {  
    name  = "id"
    type  = "S"         
  }
}




apiVersion: kubeless.io/v1beta1
kind: Function
metadata:
  name: devops-lambda
  namespace: default
spec:
  runtime: nodejs14
  handler: index.handler
  function: |
    module.exports = function(event, context) {
      context.succeed('Hello World');
    };





pipeline {
    agent any
    stages {
        stage('Build') {
            steps {
                echo 'Building Lambda function package...'
                sh 'zip -r lambda.zip lambda/'
            }
        }
        stage('Deploy to Lambda') {
            steps {
                script {
                    sh 'aws lambda update-function-code --function-name devops-lambda --zip-file fileb://lambda.zip'
                }
            }
        }
    }
}
