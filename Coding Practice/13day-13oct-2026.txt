---
- name: setup nginx tith reverse proxy
  hosts: nginx-server
  become: yes
  tasks: 
    - name: install nginx
      yum: 
        name: nginx
        state: present
    - name: configure nginx as revrse proxy
      template:
        src: reverse-proxy.conf.j2
        dest: /etc/nginx/conf.d/reverse-proxy.conf
    - name: restart ngnix service
      service:
        name: nginx
        state: restarted
        enabled: yes
 






resource "aws_s3_bucket" "devops_bucket"
  bucket = "devops-bucket"
  versioning {
    enabled = true
  }
  logging {
    target_bucket = "devops-log-bucket"
    target_prefix = "logs/"
  }
}




apiVersion: networking.k8s.io/v1
kind: Ingress
metadata:
  name: devops-ingress
  namespace: devops-dev
  annotations: 
    nginx.ingress.kubernates.io/rewrite-target: /
spec: 
  rules:
    - host: devops.local
      http:
        paths:
          - path: /
            pathType: Prefix
            backend: 
              service: 
                name: devops-service
                port: 
                  number: 80
          - path: api
            pathType: Prefix
            backend: 
              service: 
                name: devops-api-service
                port:
                  number: 80
