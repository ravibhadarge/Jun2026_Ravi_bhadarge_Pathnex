---
- name: Create new user and permission
  hosts: dev-server
  become: yes
  tasks: 
    - name: create user
      user:
        name: devuser
        state: present
        shell: /bin/bash
    - name: set permission
      file: 
        path: /home/devuser
        owner: devuser
        group: devuser
        mode: '755'

  }

}
