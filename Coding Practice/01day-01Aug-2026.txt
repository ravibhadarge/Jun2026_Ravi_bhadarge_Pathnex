provider "aws" {
  region = "us-east-1"
}

VPC
resource "aws_vpc" "ravi_vpc" {
  cidr_block           = "10.0.0.0/16"
  enable_dns_hostnames = true
  enable_dns_support   = true

  tags = {
    Name = "ravi-vpc"
  }
}

Public Subnet 
resource "aws_subnet" "public_subnet" {
  vpc_id                  = aws_vpc.ravi_vpc.id
  cidr_block              = "10.0.1.0/24"
  availability_zone       = "us-east-1a"

  tags = {
    Name = "ravi-public-subnet"
  }
}

Private Subnet 
resource "aws_subnet" "private_subnet" {
  vpc_id            = aws_vpc.ravi_vpc.id
  cidr_block        = "10.0.2.0/24"
  availability_zone = "us-east-1a"

  tags = {
    Name = "ravi-private-subnet"
  }
}

Internet Gateway
resource "aws_internet_gateway" "igw" {
  vpc_id = aws_vpc.ravi_vpc.id

  tags = {
    Name = "ravi-igw"
  }
}

Public Subnet
resource "aws_route_table" "public_rt" {
  vpc_id = aws_vpc.ravi_vpc.id

  route {
    cidr_block = "0.0.0.0/0"
    gateway_id = aws_internet_gateway.igw.id
  }

  tags = {
    Name = "ravi-public-rt"
  }
}


resource "aws_route_table_association" "public_assoc" {
  subnet_id      = aws_subnet.public_subnet.id
  route_table_id = aws_route_table.public_rt.id
}

resource "aws_security_group" "ravi_sg" {
  name        = "ravi-security-group"
  description = "Network traffic rules for ravi"
  vpc_id      = aws_vpc.ravi_vpc.id


  ingress {
    from_port   = 443
    to_port     = 443
    protocol    = "tcp"
    cidr_blocks = ["0.0.0.0/0"]
  }

  
  ingress {
    from_port   = 0
    to_port     = 0
    protocol    = "-1"
    cidr_blocks = [aws_vpc.ravi_vpc.cidr_block]
  }

  
  egress {
    from_port   = 0
    to_port     = 0
    protocol    = "-1"
    cidr_blocks = ["0.0.0.0/0"]
  }
}
