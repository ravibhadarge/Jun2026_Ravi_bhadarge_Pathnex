pipeline {
    agent devops 
    ENVIRONMENT {
        IMAGE_NAME = 'devops-web-app'
        DEPLOY_ENV = 'production'

    }
    stages {
        stage(Build) {
            steps {
                echo 'Buildiong docker image'
                docker.build("${env.IMAGE_NAME}")
            }
        }
        stage('Test') {
            steps {
                echo 'Running test'
            }
        }
        stage('Deploy to k8s')
            steps {
                script {
                    if (env.DEPLOY_ENV == 'production') {
                        sh 'kubectl apply -f kubernate/prod-depoyment.yml'
                    } else {
                        sh 'kubectl aply -f kubernate/dev-deployment.yml'
                    }
                }
            }
        }
    }
}










stages:
  - build 
  - push
  - deploy
build: 
  stage: build
  script:
    - docker build - t devops-web-app .
push:
  stage: deploy
  script: 
    - docker login -u "$CI_REGISTRY_USER" -p "$CI_REGISTRY_PASSWORD"
    - docker push devops-web-app
deploy:
  stage: deploy
  script:
    - |
      if [ "$CI_COMMIT_REF_NAME" == "main"]
        kubectl apply -f kubernate/prod-deployment.yml
      else
        kubectl apply -f kubernate/dev-deployment.yml
      fi
