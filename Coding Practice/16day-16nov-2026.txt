---
- name: create EC2 instances
  hosts: localhost
  become: yes
  tasks:
    - name: Launch EC2 instance in us-east-1
      ec2_instance:
        name: devops-ec2-us-east-1
        key_name: devops-key
        region: us-east-1
        instance_type: t2.micro
        image: ami-abcd1234abcd1234
        wait: yes
    - name: Launch EC2 instance in us-west-1
      ec2_instance:
        name: devops-ec2-us-west-1
        key_name: devops-key
        region: us-west-1
        instance_type: t2.micro
        image: ami-abcd1234abcd1234
        wait: yes







resource "aws_vpc" "us_east_vpc" {
  cidr_block = "10.0.0.0/16"
  region     = "us-east-1"
}

resource "aws_vpc" "us_west_vpc" {
  cidr_block = "10.1.0.0/16"
  region     = "us-west-1"
}

resource "aws_instance" "devops_ec2_us_east" {
  ami           = "ami-abcd1234abcd1234"
  instance_type = "t2.micro"
  region        = "us-east-1"
}

resource "aws_instance" "devops_ec2_us_west" {
  ami           = "ami-abcd1234abcd1234"
  instance_type = "t2.micro"
  region        = "us-west-1"
