---
- name: Setup prometheus Node Exporter
  hosts: devops-server
  become: yes
  tasks:
    - name: install prometheus node exporter
      yum: 
        name: prometheus-node-exporter
        state: present
    - name: start prometheus node exporter service
      service: 
        name: prometheus-node-exporter
        state: started
        enabled: yes
 

resource "aws_cloudwatch_metrix_alarm" "devops_cpu_alarm" {
  alarm_name                     = "devops-cpu-utilization-high"
  comparison_operator            = "GreaterThenThreshold"
  evaluation_operator            = "1"
  metric_name                    = "CPUUtilization"
  namespace                      = "AWS/EC2"
  period                         = "60"
  statistic                      = "Average"
  threshold                      = "80"
  
  dimension = {
    InstanceId = aws_instance.devops_ec2.id
  }
  alarm_action = [aws_sns_topic.devops_alerts.arn] 
 }
 resource "aws_sns_topic" "devops_alerts" {
   name = "devops-alerts"
 }
