---
- name: setup HAProxy
  hosts: devops-server
  become: yes
  tasks: 
    - name: install HAProxy 
      yum:
        name: haproxy
        state: present
    - name: configure haproxy
      template:
        src: haproxy.cfg.j2
        dest: /etc/haproxy/haproxy.cfg
    - name: start haproxy service
      service: 
        name: haproxy
        state: started
        enable: yes

}
